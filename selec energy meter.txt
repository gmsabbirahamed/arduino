#include <Arduino.h>
#include <ModbusMaster.h>

#define RS485_RX 16
#define RS485_TX 17

#define RS485_BAUD 9600
#define MODBUS_SLAVE_ID 2

HardwareSerial RS485Serial(2);
ModbusMaster node;


// Fix for ModbusMaster library
uint16_t makeWord(uint8_t high, uint8_t low)
{
    return ((uint16_t)high << 8) | low;
}


void setup()
{
    Serial.begin(115200);

    RS485Serial.begin(
        RS485_BAUD,
        SERIAL_8N1,
        RS485_RX,
        RS485_TX
    );

    node.begin(MODBUS_SLAVE_ID, RS485Serial);

    Serial.println();
    Serial.println("Selec MFM384 TEST");
    Serial.println("==================");
}


void loop()
{
    uint8_t result;

    // Read registers 0x00 through 0x15
    // 22 registers total
    result = node.readInputRegisters(0x00, 22);

    if (result == node.ku8MBSuccess)
    {
        Serial.println("READ OK");

        for (int i = 0; i < 22; i++)
        {
            Serial.printf(
                "REG 0x%02X = %u (0x%04X)\n",
                i,
                node.getResponseBuffer(i),
                node.getResponseBuffer(i)
            );
        }
    }
    else
    {
        Serial.print("READ FAILED: 0x");
        Serial.println(result, HEX);
    }

    Serial.println("------------------");

    delay(2000);
}
